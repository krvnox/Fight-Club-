-- ==================== DISCOVERY DE REMOTES UNIVERSAL ====================
local PacketRemote = nil
local DiscoveredCombatRemotes = {}

local function ScanForCombatRemotes()
    DiscoveredCombatRemotes = {}
    
    -- 1. Busca específica por PacketRemote
    pcall(function()
        local folder = ReplicatedStorage:FindFirstChild(CONFIG.PACKET_FOLDER)
        if folder then
            local r = folder:FindFirstChild(CONFIG.REMOTE_NAME) or folder:FindFirstChild("Packet") or folder:FindFirstChildOfClass("RemoteEvent")
            if r then PacketRemote = r end
        end
    end)
    
    -- 2. Scanner Multi-Remote em todo o ReplicatedStorage e Personagem
    for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
        if obj:IsA("RemoteEvent") or obj:IsA("RemoteFunction") then
            local n = obj.Name:lower()
            if n:find("block") or n:find("guard") or n:find("parry") or n:find("defend") or n:find("combat") or n:find("action") or n:find("packet") or n:find("skill") or n:find("hit") or n:find("attack") then
                table.insert(DiscoveredCombatRemotes, obj)
                if not PacketRemote and (n:find("packet") or n == CONFIG.REMOTE_NAME:lower()) then
                    PacketRemote = obj
                end
            end
        end
    end
    
    if not PacketRemote and #DiscoveredCombatRemotes > 0 then
        PacketRemote = DiscoveredCombatRemotes[1]
    end
end

ScanForCombatRemotes()

-- ==================== GERENCIADOR DE ANIMAÇÃO DE AUTO-BLOCK (CORRIGIDO PARA NÃO TRAVAR PERNAS) ====================
local function GetOrLoadBlockAnimationTrack(hum)
    if not hum then return nil end
    
    -- 1. Procura track de guarda/bloqueio já carregada e ativa no Humanoid
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
        local n = tostring(track.Name):lower() .. " " .. tostring(track.Animation.AnimationId):lower()
        if n:find("block") or n:find("guard") or n:find("parry") or n:find("shield") or n:find("defend") then
            return track
        end
    end
    
    local char = hum.Parent
    local animObj = nil
    
    -- 2. Busca animação de bloqueio no script nativo 'Animate' do personagem
    if char then
        local animateScript = char:FindFirstChild("Animate")
        if animateScript then
            local blockFolder = animateScript:FindFirstChild("block") or animateScript:FindFirstChild("guard") or animateScript:FindFirstChild("defend")
            if blockFolder then
                animObj = blockFolder:FindFirstChildOfClass("Animation") or blockFolder
            end
        end
        
        -- 3. Procura objeto de animação oficial no próprio personagem ou nas ferramentas
        if not animObj then
            for _, desc in ipairs(char:GetDescendants()) do
                if desc:IsA("Animation") then
                    local dn = desc.Name:lower()
                    if dn:find("block") or dn:find("guard") or dn:find("parry") or dn:find("defend") then
                        animObj = desc
                        break
                    end
                end
            end
        end
    end
    
    -- 4. Busca nos pacotes de combate do ReplicatedStorage
    if not animObj then
        for _, desc in ipairs(ReplicatedStorage:GetDescendants()) do
            if desc:IsA("Animation") then
                local dn = desc.Name:lower()
                if dn:find("block") or dn:find("guard") or dn:find("parry") then
                    animObj = desc
                    break
                end
            end
        end
    end
    
    -- 5. Fallback Universal: Animação clássica de postura de guarda R6/R15
    if not animObj then
        animObj = Instance.new("Animation")
        animObj.AnimationId = (hum.RigType == Enum.HumanoidRigType.R6) and "rbxassetid://180426354" or "rbxassetid://507766388"
    end
    
    local success, track = pcall(function()
        return hum:LoadAnimation(animObj)
    end)
    
    if success and track then
        -- CORREÇÃO V6.0: Usamos Action (não Action4) com peso adaptativo para NUNCA paralisar o deslocamento das pernas
        track.Priority = Enum.AnimationPriority.Action
        track.Looped = true
        return track
    end
    
    return nil
end

-- Executa a animação visual de Auto-Block com suporte total a andar simultaneamente
local function PlayVisibleBlockAnimation(hum)
    if not hum or not CONFIG.FORCE_VISIBLE_ANIMATION then return end
    
    local track = SubaruState.ActiveBlockTrack or GetOrLoadBlockAnimationTrack(hum)
    if track then
        SubaruState.ActiveBlockTrack = track
        if not track.IsPlaying then
            -- Se estiver andando, toca com peso 0.6 para que os braços defendam mas as pernas corram livremente!
            local weight = (hum.MoveDirection.Magnitude > 0.05) and 0.65 or 1.0
            local speedMult = CONFIG.LIGHTNING_PARRY and (CONFIG.PARRY_SPEED_MULT or 2.2) or 1.0
            track:Play(0.02, weight, speedMult)
        end
    end
end

local function StopVisibleBlockAnimation(hum)
    if SubaruState.ActiveBlockTrack then
        pcall(function()
            SubaruState.ActiveBlockTrack:Stop(0.04)
        end)
        SubaruState.ActiveBlockTrack = nil
    end
    
    if hum then
        for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
            local n = tostring(track.Name):lower() .. " " .. tostring(track.Animation.AnimationId):lower()
            if n:find("block") or n:find("guard") or n:find("parry") then
                pcall(function() track:Stop(0.04) end)
            end
        end
    end
end

-- Envio Seguro Anti-Kick (Rate-Limit & Proteção contra Desconexão do Servidor)
function CombatActions.Send(actionBuffer)
    if not PacketRemote then return end
    local now = tick()
    
    if CONFIG.ANTI_KICK_SAFETY then
        if now - SubaruState.LastPacketReset > 1 then
            SubaruState.RecentPacketCount = 0
            SubaruState.LastPacketReset = now
        end
        if SubaruState.RecentPacketCount >= 26 then
            return
        end
        SubaruState.RecentPacketCount = SubaruState.RecentPacketCount + 1
    end
    
    pcall(function()
        PacketRemote:FireServer(actionBuffer)
    end)
    SubaruState.LastAction = now
    SubaruState.LastRemoteSendTime = now
end

            end
        end
    end)
    
    if targetChar then
        local tRoot = targetChar:FindFirstChild("HumanoidRootPart")
        for _, rem in ipairs(DiscoveredCombatRemotes) do
            pcall(function()
                if rem:IsA("RemoteEvent") then

                            rem:FireServer(targetChar, tRoot.CFrame, "M1")
                            rem:FireServer("Hit", targetChar, tRoot.CFrame)
                            rem:FireServer("Attack")
                        else
                            rem:FireServer("Attack")
                            rem:FireServer("M1")
                        end
                    end
                end
            end)
        end
    end
end

-- Dispara todos os remotes de defesa descobertos com payloads universais
function CombatActions.FireAllBlockRemotes(state)
    local now = tick()
    if CONFIG.ANTI_KICK_SAFETY and (now - SubaruState.LastRemoteSendTime < 0.015) then
        return
    end
    SubaruState.LastRemoteSendTime = now
    
    for _, rem in ipairs(DiscoveredCombatRemotes) do
        pcall(function()
            if rem:IsA("RemoteEvent") then
                local rn = rem.Name:lower()
                if rn:find("block") or rn:find("guard") or rn:find("parry") or rn:find("defend") then
                    rem:FireServer(state)
                    rem:FireServer("Block", state)
                    rem:FireServer("Guard", state)
                    rem:FireServer({ Action = "Block", State = state, Key = "F" })
                    rem:FireServer("F", state)
                end
            elseif rem:IsA("RemoteFunction") then
                local rn = rem.Name:lower()
                if rn:find("block") or rn:find("guard") or rn:find("parry") or rn:find("defend") then
                    rem:InvokeServer("Block", state)
                end
            end
        end)
    end
end

-- Controle Global do Estado de Defesa (Rápido, Confiável e 100% Livre para Locomoção e Joysticks Mobile)
function CombatActions.SetBlockState(state)
    SubaruState.IsBlocking = state
    
    -- [VETOR 1: VirtualInputManager - APENAS NO PC para NUNCA travar o joystick touch no mobile]
    if not IsMobileDevice and VirtualInputManager then
        pcall(function()
            VirtualInputManager:SendKeyEvent(state, Enum.KeyCode.F, false, game)
        end)
    end
    
    -- [VETOR 2: Animação de Guarda Garantida & Suave]
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            if state and CONFIG.FORCE_VISIBLE_ANIMATION then
                PlayVisibleBlockAnimation(hum)
            else
                StopVisibleBlockAnimation(hum)
            end
            
            -- Restauração Suave sem Aceleração Forçada
            if not state then
                if CONFIG.SPEED_OVERRIDE_ENABLED then
                    local targetSpeed = CONFIG.MIN_SAFE_WALKSPEED or 16
                    if hum.WalkSpeed < targetSpeed then
                        hum.WalkSpeed = targetSpeed
                    end
                end
                pcall(function()
                    hum.PlatformStand = false
                    hum.Sit = false
                end)
            end
        end
        
        -- [VETOR 3: Sincronização de Atributos Oficiais de Defesa]
        pcall(function()
            char:SetAttribute("Blocking", state)
            char:SetAttribute("Guard", state)
            char:SetAttribute("IsBlocking", state)
            char:SetAttribute("Parry", state)
            LocalPlayer:SetAttribute("Blocking", state)
        end)
    end
    
    -- [VETOR 4: Envio de Opcodes e Multi-Remotes]
    if state then
        CombatActions.Send(ACTION_BUFFERS.BLOCK_ON)
        CombatActions.FireAllBlockRemotes(true)
    else
        CombatActions.Send(ACTION_BUFFERS.BLOCK_OFF)
        CombatActions.FireAllBlockRemotes(false)
    end
end

-- ==================== AUTO-BLOCK PURO & CONSCIENTE V6.0 ====================
function CombatActions.TriggerAutoBlock(attackKey)
    if not CONFIG.AUTO_BLOCK_ENABLED then return end
    if SubaruState.IsAttacking then return end
    if SubaruState.IsBlocking or isBlockQueued then return end
    
    local now = tick()
    local minCooldown = CONFIG.GUARD_COOLDOWN or 0.30
    if now - SubaruState.LastBlockFinishedTime < minCooldown then return end
    if now - SubaruState.LastBlockTime < minCooldown then return end
    
    if attackKey and SubaruState.HandledAttacks[attackKey] and (now - SubaruState.HandledAttacks[attackKey] < 0.65) then
        return
    end
    if attackKey then SubaruState.HandledAttacks[attackKey] = now end
    
    SubaruState.LastBlockTime = now
    isBlockQueued = true
    
    local blockHoldDuration = CONFIG.FAST_BLOCK_DURATION or 0.065
    local reactionDelay = 0.002
    
    if CONFIG.HUMANIZE_BLOCK then
        local jitter = (math.random(-5, 10) / 1000)
        reactionDelay = math.max(0.02, (CONFIG.HUMAN_REACTION_DELAY or 0.08) + jitter)
        blockHoldDuration = math.max(0.08, (CONFIG.HUMAN_HOLD_DURATION or 0.12) + (math.random(-5, 5) / 1000))
    else
        blockHoldDuration = CONFIG.LIGHTNING_PARRY and 0.045 or (CONFIG.FAST_BLOCK_DURATION or 0.065)
    end
    
    task.delay(reactionDelay, function()
        isBlockQueued = false
        if not SubaruState.Enabled or SubaruState.IsAttacking then return end
        
        CombatActions.SetBlockState(true)
        
        if SubaruState.BlockThread then task.cancel(SubaruState.BlockThread) end
        
        SubaruState.BlockThread = task.delay(blockHoldDuration, function()
            CombatActions.ReleaseBlock()
        end)
    end)
end

-- Liberação Garantida de Guarda e Restauração Total de Movimento
function CombatActions.ReleaseBlock()
    isBlockQueued = false
    if SubaruState.BlockThread then
        task.cancel(SubaruState.BlockThread)
        SubaruState.BlockThread = nil
    end
    SubaruState.PriorityMode = false
    SubaruState.LastBlockFinishedTime = tick()
    CombatActions.SetBlockState(false)
    
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            if CONFIG.SPEED_OVERRIDE_ENABLED then
                local targetSpeed = CONFIG.MIN_SAFE_WALKSPEED or 16
                if hum.WalkSpeed < targetSpeed then
                    hum.WalkSpeed = targetSpeed
                end
            end
            pcall(function()
                hum.PlatformStand = false
                hum.Sit = false
            end)
        end
    end
end

-- Executa Quebra de Guarda Instantânea (Anti-Block)
function CombatActions.ExecuteGuardBreak()
    local now = tick()
    if now - SubaruState.LastGuardBreakTime < 0.12 then return end
    SubaruState.LastGuardBreakTime = now
    
    CombatActions.PerformLowKick(SubaruState.Target)
    
    task.delay(0.06, function()
        CombatActions.PerformM1Attack(SubaruState.Target)
    end)
end

-- ==================== SHIELD PIERCER (BYPASS DE DEFESA DO ADVERSÁRIO) ====================
function CombatActions.SendWithFlankBypass(actionBuffer, targetChar)
    if not targetChar then
        if actionBuffer == ACTION_BUFFERS.HIGH_KICK then
            CombatActions.PerformHighKick(nil)
        elseif actionBuffer == ACTION_BUFFERS.LOW_KICK then
            CombatActions.PerformLowKick(nil)
        else
            CombatActions.PerformM1Attack(nil)
        end
        return
    end

    local isEnemyBlocking = false
    if targetChar:GetAttribute("Blocking") or targetChar:GetAttribute("Guard") or targetChar:GetAttribute("IsBlocking") then
        isEnemyBlocking = true
    end
    local bVal = targetChar:FindFirstChild("Blocking") or targetChar:FindFirstChild("Guard")
    if bVal and (bVal.Value == true or bVal.Value == 1) then
        isEnemyBlocking = true
    end
    
    local tHum = targetChar:FindFirstChildOfClass("Humanoid")
    if tHum then
        for _, tr in ipairs(tHum:GetPlayingAnimationTracks()) do
            local tn = tostring(tr.Name):lower() .. " " .. tostring(tr.Animation.AnimationId):lower()
            if tn:find("block") or tn:find("guard") or tn:find("parry") or tn:find("shield") then
                isEnemyBlocking = true
                break
            end
        end
    end

    local tRoot = targetChar:FindFirstChild("HumanoidRootPart")
    
    if isEnemyBlocking and CONFIG.SHIELD_PIERCER then
        if tRoot and CONFIG.ANGLE_BYPASS then
            pcall(function()
                local flankCFrame = tRoot.CFrame * CFrame.Angles(0, math.rad(180), 0)
                for _, rem in ipairs(DiscoveredCombatRemotes) do
                    if rem:IsA("RemoteEvent") then
                        rem:FireServer("Hit", targetChar, flankCFrame, actionBuffer)
                        rem:FireServer("M1", targetChar, flankCFrame)
                        rem:FireServer("Damage", targetChar, flankCFrame)
                        rem:FireServer("Kick", targetChar, flankCFrame)
                    end
                end
            end)
        end

        if actionBuffer == ACTION_BUFFERS.HIGH_KICK then
            CombatActions.PerformHighKick(targetChar)
        elseif actionBuffer == ACTION_BUFFERS.LOW_KICK then
            CombatActions.PerformLowKick(targetChar)
        else
            CombatActions.PerformM1Attack(targetChar)
        end
        
        if CONFIG.ANTI_BLOCK_SPAM then
            task.spawn(function()
                CombatActions.ExecuteGuardBreak()
            end)
        end
    else
        if actionBuffer == ACTION_BUFFERS.HIGH_KICK then
            CombatActions.PerformHighKick(targetChar)
        elseif actionBuffer == ACTION_BUFFERS.LOW_KICK then
            CombatActions.PerformLowKick(targetChar)
        else
            CombatActions.PerformM1Attack(targetChar)
        end
    end
end

-- ==================== MÓDULO GAME SENSE (INTELIGÊNCIA & CONSCIÊNCIA DE JOGO V6.0) ====================
local GameSense = {}

function GameSense.AnalyzeTarget(targetChar)
    local analysis = {
        isValid = false,
        distance = 999,
        isFacingUs = false,
        isInMeleeReach = false,
        isAttacking = false,
        attackKey = nil,
        isGuardBreak = false,
        isWhiffRecovery = false,
        isKnockedDownIFrame = false,
        isStunnedOrRagdoll = false,
        isBlocking = false,
        healthPercent = 100,
        threatVerdict = "SAFE"
    }

    if not targetChar or not targetChar:FindFirstChild("HumanoidRootPart") then return analysis end
    local pChar = LocalPlayer.Character
    if not pChar or not pChar:FindFirstChild("HumanoidRootPart") then return analysis end

    local pRoot = pChar.HumanoidRootPart
    local tRoot = targetChar.HumanoidRootPart
    local tHum = targetChar:FindFirstChildOfClass("Humanoid")

    if not tHum or tHum.Health <= 0 then return analysis end

    analysis.isValid = true
    analysis.distance = (pRoot.Position - tRoot.Position).Magnitude
    analysis.healthPercent = (tHum.Health / tHum.MaxHealth) * 100

    local toPlayer = (pRoot.Position - tRoot.Position).Unit
    local dotProduct = tRoot.CFrame.LookVector:Dot(toPlayer)
    analysis.isFacingUs = (dotProduct > -0.2) or (analysis.distance <= (CONFIG.MAX_MELEE_REACH or 7.2))

    local maxReach = math.max(CONFIG.MAX_MELEE_REACH or 7.2, 7.2)
    analysis.isInMeleeReach = (analysis.distance <= maxReach)
    analysis.isSpacingDistance = (analysis.distance > maxReach and analysis.distance <= (CONFIG.SPACING_MAX_REACH or 13.5))
    
    local tVel = tRoot.AssemblyLinearVelocity or tRoot.Velocity or Vector3.zero
    local approachVelocity = tVel:Dot(toPlayer)
    analysis.isLungingTowardsUs = (approachVelocity > 6.0)
    analysis.approachVelocity = approachVelocity

    if targetChar:GetAttribute("Ragdoll") == true or targetChar:GetAttribute("KnockedDown") == true or targetChar:GetAttribute("Stunned") == true or targetChar:GetAttribute("Invulnerable") == true then
        analysis.isKnockedDownIFrame = true
        analysis.isStunnedOrRagdoll = true
    end

    if targetChar:GetAttribute("Blocking") == true or targetChar:GetAttribute("Guard") == true or targetChar:GetAttribute("IsBlocking") == true then
        analysis.isBlocking = true
    end
    local bVal = targetChar:FindFirstChild("Blocking") or targetChar:FindFirstChild("Guard")
    if bVal and (bVal.Value == true or bVal.Value == 1) then
        analysis.isBlocking = true
    end

    if targetChar:GetAttribute("Attacking") == true or targetChar:GetAttribute("Swing") == true or targetChar:GetAttribute("Combat") == true or targetChar:GetAttribute("M1") == true then
        analysis.isAttacking = true
        analysis.attackKey = "ATTR_ATTACK_" .. tostring(targetChar) .. "_" .. tostring(math.floor(tick() * 3))
    end

    local tracks = {}
    pcall(function()
        for _, tr in ipairs(tHum:GetPlayingAnimationTracks()) do
            table.insert(tracks, tr)
        end
    end)
    local animator = tHum:FindFirstChildOfClass("Animator")
    if animator then
        pcall(function()
            for _, tr in ipairs(animator:GetPlayingAnimationTracks()) do
                table.insert(tracks, tr)
            end
        end)
    end

    for _, track in ipairs(tracks) do
        if track.IsPlaying and track.WeightCurrent > 0.15 then
            local aId = tostring(track.Animation.AnimationId):lower()
            local tName = tostring(track.Name):lower()
            local combined = aId .. " " .. tName

            if combined:find("block") or combined:find("guard") or combined:find("parry") or combined:find("shield") then
                analysis.isBlocking = true
            end

            if combined:find("heavy") or combined:find("break") or combined:find("guardbreak") or combined:find("smash") or combined:find("charge") then
                analysis.isGuardBreak = true
            end

            local isLocomotion = (combined:find("idle") or combined:find("walk") or combined:find("run") or combined:find("sprint") or combined:find("jump") or combined:find("fall") or combined:find("land") or combined:find("dance") or combined:find("emote") or combined:find("climb") or combined:find("swim"))

            if not isLocomotion then
                local hasAttackKeyword = (combined:find("attack") or combined:find("punch") or combined:find("slash") or combined:find("hit") or combined:find("swing") or combined:find("kick") or combined:find("m1") or combined:find("m2") or combined:find("combo") or combined:find("strike") or combined:find("jab") or combined:find("hook") or combined:find("skill") or combined:find("blade") or combined:find("sword") or combined:find("katana"))

                if hasAttackKeyword then
                    local totalLen = track.Length > 0 and track.Length or 0.8
                    local progress = track.TimePosition / totalLen

                    if progress >= 0.04 and progress <= 0.75 then
                        analysis.isAttacking = true
                        local swingCycle = math.floor(track.TimePosition * 4)
                        analysis.attackKey = tostring(track.Animation.AnimationId) .. "_cycle_" .. tostring(swingCycle)
                    elseif progress > 0.75 then
                        analysis.isWhiffRecovery = true
                    end
                end
            end

            if combined:find("fall") or combined:find("down") or combined:find("ragdoll") or combined:find("getup") or combined:find("wakeup") then
                analysis.isKnockedDownIFrame = true
            end
        end
    end

    return analysis
end

function GameSense.ShouldBlock(analysis)
    if not CONFIG.AUTO_BLOCK_ENABLED then return false, "AUTO_BLOCK_DISABLED" end
    if SubaruState.IsAttacking then return false, "PLAYER_IS_ATTACKING" end

    if analysis.isGuardBreak and CONFIG.ANTI_GB_EVASION then
        return false, "ENEMY_CHARGING_GUARD_BREAK"
    end

    if not analysis.isFacingUs then
        return false, "ENEMY_NOT_FACING_US"
    end
    
    local maxReach = CONFIG.MAX_MELEE_REACH or 7.2
    if analysis.distance > maxReach then
        return false, "OUT_OF_MELEE_REACH"
    end

    if analysis.isStunnedOrRagdoll or analysis.isKnockedDownIFrame then
        return false, "ENEMY_STUNNED_OR_DOWN"
    end

    if analysis.isWhiffRecovery then
        return false, "ENEMY_IN_WHIFF_RECOVERY"
    end

    if analysis.isBlocking then
        return false, "ENEMY_IS_BLOCKING"
    end

    if tick() - SubaruState.LastBlockFinishedTime < (CONFIG.GUARD_COOLDOWN or 0.30) then
        return false, "GUARD_COOLDOWN_ACTIVE"
    end

    if analysis.isAttacking and analysis.distance <= maxReach then
        return true, "REAL_HITBOX_THREAT"
    end

    return false, "NO_THREAT"
end

function GameSense.GetOptimalTacticalAction(analysis)
    if not analysis.isValid then return "IDLE" end

    if analysis.isBlocking and CONFIG.ANTI_BLOCK_ENABLED then
        return "EXECUTE_GUARD_BREAK"
    end

    if analysis.isGuardBreak and CONFIG.ANTI_GB_EVASION and analysis.isInMeleeReach then
        return "INTERRUPT_GUARD_BREAK"
    end

    if analysis.isKnockedDownIFrame and CONFIG.OKIZEME_WAKEUP_GUARD then
        return "WAIT_WAKEUP_TIMING"
    end

    if analysis.isWhiffRecovery and CONFIG.WHIFF_PUNISH_ENABLED and analysis.distance <= CONFIG.ATTACK_RANGE then
        return "WHIFF_PUNISH"
    end

    if analysis.healthPercent <= 30 and CONFIG.BURST_DPS_LOW_HP and analysis.distance <= CONFIG.ATTACK_RANGE then
        return "BURST_FINISHER"
    end

    if analysis.distance <= CONFIG.ATTACK_RANGE then
        return "STANDARD_COMBO"
    end

    return "APPROACHING"
end

local function ExecuteSmartCombo(tacticalAction)
    if not SubaruState.Enabled or not CONFIG.AUTO_COMBO_ENABLED then return end
    if SubaruState.IsAttacking or SubaruState.IsBlocking then return end
    if tick() - SubaruState.LastComboTime < CONFIG.COMBO_DELAY then return end
    
    SubaruState.IsAttacking = true
    SubaruState.LastComboTime = tick()
    
    task.spawn(function()
        local pattern = EnhancedPatterns[1]
        
        if tacticalAction == "WHIFF_PUNISH" then
            pattern = { ACTION_BUFFERS.PUNCH, 0.065, ACTION_BUFFERS.PUNCH, 0.07, ACTION_BUFFERS.HIGH_KICK, 0.09 }
        elseif tacticalAction == "BURST_FINISHER" then
            pattern = { ACTION_BUFFERS.PUNCH, 0.065, ACTION_BUFFERS.HIGH_KICK, 0.08 }
        elseif tacticalAction == "INTERRUPT_GUARD_BREAK" then
            pattern = { ACTION_BUFFERS.PUNCH, 0.06, ACTION_BUFFERS.PUNCH, 0.065, ACTION_BUFFERS.LOW_KICK, 0.08 }
        else
            pattern = EnhancedPatterns[math.random(1, #EnhancedPatterns)]
        end
        
        for i = 1, #pattern, 2 do
            if not SubaruState.Target or not SubaruState.Target:FindFirstChild("Humanoid") or SubaruState.Target.Humanoid.Health <= 0 then
                break
            end
            
            if SubaruState.PriorityMode or SubaruState.IsBlocking then break end
            
            local liveAnalysis = GameSense.AnalyzeTarget(SubaruState.Target)
            if SubaruState.AntiBlock and liveAnalysis.isBlocking then
                CombatActions.ExecuteGuardBreak()
                task.wait(0.07)
            end
            
            local move = pattern[i]
            local delayTime = pattern[i+1] or CONFIG.PERFECT_M1_SPEED
            
            CombatActions.SendWithFlankBypass(move, SubaruState.Target)
            task.wait(delayTime)
        end
        
        SubaruState.IsAttacking = false
    end)
end

-- ==================== EVENT-DRIVEN AUTO-BLOCK ENGINE ====================
local MonitoredEntities = {}

local function HookEntityForAutoBlock(entityModel)
    if not entityModel or entityModel == LocalPlayer.Character then return end
    if MonitoredEntities[entityModel] then return end
    
    local hum = entityModel:FindFirstChildOfClass("Humanoid")
    local root = entityModel:FindFirstChild("HumanoidRootPart") or entityModel:FindFirstChild("Torso") or entityModel:FindFirstChild("UpperTorso")
    if not hum or not root then return end
    
    MonitoredEntities[entityModel] = true
    
    local function OnAnimationTriggered(track)
        if not SubaruState.Enabled or not CONFIG.AUTO_BLOCK_ENABLED then return end
        if SubaruState.IsAttacking or SubaruState.IsBlocking or isBlockQueued then return end
        if (tick() - SubaruState.LastBlockFinishedTime < (CONFIG.GUARD_COOLDOWN or 0.30)) then return end
        
        local pChar = LocalPlayer.Character
        if not pChar or not pChar:FindFirstChild("HumanoidRootPart") then return end
        
        local pRoot = pChar.HumanoidRootPart
        local dist = (pRoot.Position - root.Position).Magnitude
        local maxReach = CONFIG.MAX_MELEE_REACH or 7.2
        if dist > maxReach then return end
        
        local toUs = (pRoot.Position - root.Position).Unit
        local lookDot = root.CFrame.LookVector:Dot(toUs)
        if lookDot < 0.20 then return end
        
        local animName = tostring(track.Name):lower()
        local animId = tostring(track.Animation.AnimationId):lower()
        local combined = animName .. " " .. animId
        
        if combined:find("idle") or combined:find("walk") or combined:find("run") or combined:find("sprint") or combined:find("jump") or combined:find("fall") or combined:find("land") or combined:find("dance") or combined:find("emote") or combined:find("climb") or combined:find("swim") then
            return
        end
        
        if combined:find("block") or combined:find("guard") or combined:find("parry") or combined:find("down") or combined:find("ragdoll") then
            return
        end
        
        local isAttack = (combined:find("attack") or combined:find("punch") or combined:find("slash") or combined:find("hit") or combined:find("swing") or combined:find("kick") or combined:find("m1") or combined:find("m2") or combined:find("strike") or combined:find("blade") or combined:find("sword") or combined:find("katana") or combined:find("jab") or combined:find("hook") or combined:find("combo"))
        
        if isAttack then
            local attackKey = animId .. "_" .. tostring(entityModel) .. "_" .. tostring(math.floor(tick() * 1.5))
            CombatActions.TriggerAutoBlock(attackKey)
        end
    end
    
    local animator = hum:FindFirstChildOfClass("Animator")
    if animator then
        animator.AnimationPlayed:Connect(OnAnimationTriggered)
    else
        hum.AnimationPlayed:Connect(OnAnimationTriggered)
    end
end

            CombatActions.ReleaseBlock()

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local enemyHum = p.Character:FindFirstChild("Humanoid")
            if enemyHum and enemyHum.Health > 0 then
                HookEntityForAutoBlock(p.Character)
                local d = (char.HumanoidRootPart.Position - p.Character.HumanoidRootPart.Position).Magnitude
                if d < shortestDist then
                    shortestDist = d
                    closestEnemy = p.Character
                end
            end
        end
    end
    
    for _, obj in ipairs(workspace:GetChildren()) do
        if obj:IsA("Model") and obj ~= char and obj:FindFirstChild("HumanoidRootPart") and obj:FindFirstChildOfClass("Humanoid") then
            local enemyHum = obj:FindFirstChildOfClass("Humanoid")
            if enemyHum.Health > 0 then
                HookEntityForAutoBlock(obj)
                local d = (char.HumanoidRootPart.Position - obj.HumanoidRootPart.Position).Magnitude
                if d < shortestDist then
                    shortestDist = d
                    closestEnemy = obj
                end
            end
        end
    end

        local analysis = GameSense.AnalyzeTarget(closestEnemy)

                CombatActions.ReleaseBlock()
            end
            
            if CONFIG.AUTO_COMBO_ENABLED then
                if tacticalAction == "EXECUTE_GUARD_BREAK" then

                elseif tacticalAction == "WAIT_WAKEUP_TIMING" then

                    if SubaruState.IsBlocking then
                        CombatActions.ReleaseBlock()
                    end
                    ExecuteSmartCombo(tacticalAction)
                end
            end
        end
    else
        if SubaruState.IsBlocking and (tick() - SubaruState.LastBlockTime > 0.05) then
            CombatActions.ReleaseBlock()
        end
        SubaruState.EnemyState = "NEUTRAL"
    end

            frameDragStart = input.Position
            frameStartPos = MainFrame.Position
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    draggingFrame = false
                end
            end)
        end
    end)

        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            if draggingFrame and frameDragStart and frameStartPos then
                local delta = input.Position - frameDragStart
                MainFrame.Position = UDim2.new(
                    frameStartPos.X.Scale,
                    frameStartPos.X.Offset + delta.X,
                    frameStartPos.Y.Scale,
                    frameStartPos.Y.Offset + delta.Y
                )
            end
        end
    end)
